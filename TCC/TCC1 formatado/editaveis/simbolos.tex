% Símbolos usados no Capítulo 4 (Algoritmo 1 e partições).
\begin{simbolos}
  \item[$\mathcal{A}$] Conjunto de modelos de fundação avaliados
  \item[$a$] Modelo de fundação
  \item[$B$] Número de reamostragens no \textit{bootstrap}
  \item[$C$] Coleção curada
  \item[$D$] Conjunto de desenvolvimento
  \item[$F_i$] Dobra de validação $i$
  \item[$\mathcal{H}_a$] Grade de hiperparâmetros do modelo $a$
  \item[$k$] Número de dobras da validação cruzada
  \item[$n$] Número de casos usado no cálculo de um intervalo
  \item[$p$] Proporção estimada, como a sensibilidade
  \item[$S$] Conjunto de sementes
  \item[$T$] Teste reservado
  \item[$\tau$] Limiar de decisão
\end{simbolos}
